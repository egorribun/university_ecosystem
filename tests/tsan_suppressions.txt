# TSan suppressions file for Python FFI tests.
#
# WHY: TSan instruments every memory access. When our Rust extension is loaded
# alongside CPython (which uses libuv as its asyncio event loop) and tokio
# (which uses its own thread pool), TSan reports data races inside libuv's
# internal pipe/signal dispatch code. These are NOT races in our code — they
# are known, intentional patterns in libuv's design where two event-loop
# threads share a pipe fd using separate read/write ends.
#
# IMPORTANT — why called_from_lib instead of race:<function>:
#   On the CI runner, libuv is installed from the Ubuntu apt package without
#   debug symbols. TSan therefore shows all libuv frames as:
#       <null> <null> (uv+0x1234567)
#   Function-name suppressions (race:uv__io_poll) never fire because TSan
#   cannot resolve the symbol. The called_from_lib: pattern matches by the
#   DSO path, which is always available regardless of symbol visibility.

# uvloop statically bundles libuv into its extension module, so the DSO is
# named uvloop rather than libuv on the CI runner. This is intentionally the
# only DSO-wide allowance: the extension is an explicit, pinned dependency of
# the test environment, and the canary in run_tsan_tests.sh proves that broad
# suppression cannot make the gate pass by itself.
called_from_lib:uvloop

# ── tokio / Mio epoll wakeup pipe ───────────────────────────────────────────
# tokio's mio backend uses the same self-pipe wakeup trick as libuv.
# Mio is statically linked into the Rust extension, so its symbols ARE
# visible — function-name suppression works here.
race:mio::sys::unix::waker
race:mio::sys::unix::pipe

# ── libuv function names (for runners that DO have debug symbols) ────────────
# Belt-and-suspenders: if debug symbols are ever present, these fire instead.
race:uv__io_poll
race:uv__async_send
race:uv__async_io
race:uv__signal_handler
race:uv__write
race:uv__read
race:uv_run
race:uv__stream_io
race:uv__pipe_connect
race:uv__tcp_connect
race:uv_pipe_open
race:uv_async_send

# ── CPython function names (for runners with Python debug symbols) ────────────
race:_PyEval_SignalAsyncExc
race:_PyEval_SignalReceived
race:PyEval_SaveThread
race:PyEval_RestoreThread
race:take_gil
race:drop_gil
race:_Py_atomic_load_relaxed_impl
race:_PyRuntime
race:t_bootstrap
race:thread_PyThread_start_new_thread

# ── PyO3 free-threaded argument adapters ─────────────────────────────────────
# PyO3 0.29's generated argument converters for the CPython 3.14 free-threaded
# ABI create/borrow temporary argument storage before the Rust function body is
# entered. TSan reports those adapter accesses as races when the same pure,
# immutable Python inputs are passed concurrently. The typed Rust cores below
# receive owned values and remain instrumented; only the generated adapters are
# suppressed.
race:__pyfunction_batch_detect_conflicts_py
race:__pyfunction_verify_audit_signature
